{{- $alt := .Text | plainify -}} {{- $resource := .Page.Resources.GetMatch .Destination -}} {{- if $resource -}} {{- if eq $resource.MediaType.SubType "gif" -}} {{/* GIF 不转码，否则动图会变成静态帧。 */}} {{- else -}} {{/* 三档宽度 + srcset：手机不再下 1400px 整图，宽屏仍拿高清版。 24px 低质占位以内联 base64 铺满图框，解码完成后被真图覆盖， 慢网下看到的是模糊预览而不是长时间白框。 */}} {{- $srcset := slice -}} {{- $largest := false -}} {{- range (slice 600 1000 1400) -}} {{- if ge $resource.Width . -}} {{- $variant := $resource.Resize (printf "%dx webp q78" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $variant.RelPermalink $variant.Width) -}} {{- $largest = $variant -}} {{- end -}} {{- end -}} {{- if not $largest -}} {{- $largest = $resource.Process "webp q78" -}} {{- end -}} {{- $lqip := $resource.Fit "24x24 webp q25" -}} {{- $lqipURL := printf "data:image/webp;base64,%s" ($lqip.Content | base64Encode) -}} {{/* 整个 style 属性走 safeHTMLAttr：走 safeCSS 的话 html/template 仍会把 base64 里的 引号/加号转成实体（虽然能解析回来，但输出很难看，也容易在个别浏览器翻车）。 不加引号的 url() 里 + / = 都是合法字符，所以直接拼 data URI。 */}} {{- $lqipAttr := printf `style="background-image:url(%s);background-size:100%% 100%%;background-repeat:no-repeat"` $lqipURL -}} {{- end -}} {{- else -}} {{- end -}}
