{{/* Topic chips must point at tags that actually exist: the hand-written list in data/sidebar.yaml drifts from what the posts use (it once contained Protocol / Web / Tooling, none of which any post carried), and a chip leading to an empty search reads as a broken site. Counting is case-insensitive because the archive mixes "CTF" and "ctf" — as two separate keys they looked like 18 posts each and the chip disagreed with the search result it opened. The first spelling seen is the label. */}} {{- $posts := where site.RegularPages "Section" "posts" -}} {{- $counts := dict -}} {{- $labels := dict -}} {{- range $posts -}} {{- $seen := slice -}} {{- range (.Params.tags | default slice) -}} {{- $key := lower . -}} {{/* Some posts list both "CTF" and "ctf"; counting each occurrence made the chip claim 44 posts on a 38-post blog. One post counts once per tag. */}} {{- if not (in $seen $key) -}} {{- $seen = $seen | append $key -}} {{- $counts = merge $counts (dict $key (add (index $counts $key | default 0) 1)) -}} {{- if not (index $labels $key) -}} {{- $labels = merge $labels (dict $key .) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $chosen := slice -}} {{- range .items -}} {{- $key := lower . -}} {{- if and (index $counts $key) (not (in $chosen $key)) -}} {{- $chosen = $chosen | append $key -}} {{- end -}} {{- end -}} {{- if lt (len $chosen) 6 -}} {{- $pairs := slice -}} {{- range $key, $count := $counts -}} {{- $pairs = $pairs | append (dict "key" $key "count" $count) -}} {{- end -}} {{- range sort $pairs "count" "desc" -}} {{- if and (lt (len $chosen) 6) (not (in $chosen .key)) -}} {{- $chosen = $chosen | append .key -}} {{- end -}} {{- end -}} {{- end -}} {{/* Sort what survived by usage so the row reads as "what this blog is actually about" instead of the order someone typed into sidebar.yaml months ago. */}} {{- $final := slice -}} {{- range $chosen -}} {{- $final = $final | append (dict "key" . "count" (index $counts . | default 0)) -}} {{- end -}} {{- $final = first 8 (sort $final "count" "desc") -}} {{/* Phones scroll the chips on one line; tablets and up wrap them. */}}
