{{- /* Breadcrumb trail. items: slice of dicts { label, url }. The last item is rendered as the current page (no link) when it has no url, so callers only pass urls for the ancestors. */ -}} {{- $items := .items | default slice -}} {{- if $items }}
{{- range $index, $item := $items }} {{- if gt $index 0 }} / {{- end }} {{- if $item.url }} {{ $item.label }} {{- else }} {{ $item.label }} {{- end }} {{- end }}
{{- end }}
